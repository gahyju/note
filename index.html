} else { selectedCauses.add("문제 조건 해석 누락"); } renderCauseChips(); // 결과 배너 표시 const banner = document.getElementById('ai-prediction-banner'); const titleEl = document.getElementById('ai-pred-title'); reasonEl.textContent = `개념 이해도 ${Math.round(mastery*100)}%, 권장 시간 대비 ${timeRatio.toFixed(1)}배 소요 패턴을 분석해 자동 분류했습니다.`; verEl.textContent = `내 데이터 ${problems.length}건 학습 반영됨`; showToast(`AI가 '${DIAGNOSIS_META[bestType].label}' 유형으로 자동 분류했습니다!`, "success");
